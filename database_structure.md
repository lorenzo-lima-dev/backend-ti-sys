'''sql
CREATE TABLE "TECNICOS"."TECNICOS" (
    id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    nome VARCHAR(100) NOT NULL DEFAULT 'Técnico',
    cargo VARCHAR(100) NOT NULL DEFAULT 'Técnico de TI',
    criado_em TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
'''

'''sql
CREATE TABLE "TECNICOS"."LOGINS" (
    id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    id_tecnico BIGINT REFERENCES "TECNICOS"."TECNICOS"(id),
    email VARCHAR(100) NOT NULL UNIQUE,
    email_secundario VARCHAR(100) NOT NULL UNIQUE,
    senha_hash TEXT NOT NULL,
    criado_em TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
'''
